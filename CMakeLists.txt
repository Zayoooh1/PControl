cmake_minimum_required(VERSION 3.24)
project(PControl VERSION 0.1.0 LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "PControl requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz URL_HASH SHA256=d6c65aca6b1ed68e7a182f4757257b107ae403032760ed6ef121c9d55e81757d)
FetchContent_MakeAvailable(json)
add_library(shared src/shared.cpp src/ipc.cpp)
target_include_directories(shared PUBLIC src)
target_link_libraries(shared PUBLIC nlohmann_json::nlohmann_json advapi32 shell32)
target_compile_definitions(shared PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
add_library(engine src/process.cpp src/config.cpp src/logger.cpp src/rules.cpp src/engine.cpp)
target_link_libraries(engine PUBLIC shared)
add_executable(PControl.Core WIN32 src/core.cpp)
target_link_libraries(PControl.Core PRIVATE engine)
add_executable(PControl WIN32 src/gui.cpp)
target_link_libraries(PControl PRIVATE shared comctl32)
add_executable(PControl.Tests tests/tests.cpp)
target_link_libraries(PControl.Tests PRIVATE engine)
add_executable(PControl.TestTarget tests/target.cpp)
target_link_libraries(PControl.TestTarget PRIVATE shared)
foreach(t shared engine PControl.Core PControl PControl.Tests PControl.TestTarget)
  target_compile_options(${t} PRIVATE /W4 /permissive- /utf-8)
endforeach()
enable_testing()
add_test(NAME stage1 COMMAND PControl.Tests)
